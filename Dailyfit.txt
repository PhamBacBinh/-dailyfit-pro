    <script>
        const EXS = {
            "1": { name: "Chống đẩy (CĐ)", unit: "cái", met: 8.0, timePerRep: 3.0 },
            "2": { name: "Gập bụng (GB)", unit: "cái", met: 3.8, timePerRep: 2.5 },
            "3": { name: "Squat (SQ)", unit: "cái", met: 5.0, timePerRep: 3.0 },
            "4": { name: "Xà đơn (XĐ)", unit: "cái", met: 8.0, timePerRep: 3.5 },
            "5": { name: "Chạy bộ (CB)", unit: "phút", met: 11.5, timePerRep: 60.0 }
        };
        const WEIGHT = 78;
        let logs = JSON.parse(localStorage.getItem('dailyfit_logs') || '[]');
        let currentEx = "1";
        let stopwatchInterval = null, secondsElapsed = 0, isRunning = false;

        function init() { renderExercises(); updateDashboard(); }

        function renderExercises() {
            const container = document.getElementById('exerciseList');
            container.innerHTML = Object.keys(EXS).map(k => `
                <div onclick="selectExercise('${k}')" class="p-3 rounded-xl border ${currentEx === k ? 'border-cyan-500 bg-cyan-500/10' : 'border-darkborder bg-darkbg'} flex justify-between items-center cursor-pointer">
                    <div>
                        <div class="font-bold text-sm">${EXS[k].name}</div>
                        <div class="text-[10px] text-slate-400">Đơn vị: ${EXS[k].unit}</div>
                    </div>
                    <div class="flex items-center gap-2">
                        <input type="number" id="input_${k}" value="10" onclick="event.stopPropagation()" class="w-16 bg-darkcard border border-darkborder text-center py-1.5 rounded-lg text-xs font-bold text-cyan-400">
                        <button onclick="event.stopPropagation(); logExercise('${k}')" class="bg-cyan-500 text-darkbg px-3 py-1.5 rounded-lg font-bold text-xs">Lưu</button>
                    </div>
                </div>
            `).join('');
        }

        function selectExercise(k) {
            currentEx = k; renderExercises();
            if(k === "5") {
                document.getElementById('timerCard').classList.remove('hidden');
                document.getElementById('activeTimerTitle').innerText = "ĐỒNG HỒ " + EXS[k].name.toUpperCase();
            } else {
                document.getElementById('timerCard').classList.add('hidden'); pauseStopwatch();
            }
        }

        function calculateCals(code, amount) {
            const ex = EXS[code];
            const calPerMin = (ex.met * 3.5 * WEIGHT) / 200;
            return ex.unit === 'cái' ? calPerMin * (ex.timePerRep / 60.0) * amount : calPerMin * amount;
        }

        function logExercise(code, customAmount = null) {
            const amount = customAmount !== null ? customAmount : parseFloat(document.getElementById(`input_${code}`).value) || 0;
            if(amount <= 0) return;
            const cals = calculateCals(code, amount);
            const today = new Date().toISOString().split('T')[0];
            logs.push({ date: today, code, amount, cals, time: new Date().toLocaleTimeString() });
            localStorage.setItem('dailyfit_logs', JSON.stringify(logs));
            updateDashboard();
        }

        function updateDashboard() {
            const today = new Date().toISOString().split('T')[0];
            const todayLogs = logs.filter(l => l.date === today);
            const totalReps = todayLogs.reduce((sum, l) => sum + l.amount, 0);
            const todayCals = todayLogs.reduce((sum, l) => sum + l.cals, 0);
            const allCals = logs.reduce((sum, l) => sum + l.cals, 0);

            document.getElementById('todayReps').innerText = totalReps + " lượt";
            document.getElementById('todayCals').innerText = todayCals.toFixed(1) + " kcal";
            document.getElementById('totalCals').innerText = Math.round(allCals) + " kcal";

            document.getElementById('todayHistoryList').innerHTML = todayLogs.length === 0 ? 
                '<div class="text-slate-500 text-center py-2">Chưa có hiệp tập hôm nay</div>' :
                todayLogs.map((l, idx) => `
                    <div class="flex justify-between items-center bg-darkbg p-2 rounded-lg border border-darkborder">
                        <div><span class="font-bold text-cyan-400">${EXS[l.code].name}</span>: ${l.amount} ${EXS[l.code].unit} <span class="text-slate-500">(${l.time})</span></div>
                        <div class="flex items-center gap-2"><span class="text-amber-400">+${l.cals.toFixed(1)} kcal</span><button onclick="deleteLog(${idx})" class="text-rose-400"><i class="fa-solid fa-trash"></i></button></div>
                    </div>
                `).join('');
        }

        function deleteLog(idx) {
            const today = new Date().toISOString().split('T')[0];
            let todayIdx = 0;
            for(let i=0; i<logs.length; i++) {
                if(logs[i].date === today) {
                    if(todayIdx === idx) { logs.splice(i, 1); break; }
                    todayIdx++;
                }
            }
            localStorage.setItem('dailyfit_logs', JSON.stringify(logs));
            updateDashboard();
        }

        function clearToday() {
            if(!confirm("Xóa tất cả hiệp tập hôm nay?")) return;
            const today = new Date().toISOString().split('T')[0];
            logs = logs.filter(l => l.date !== today);
            localStorage.setItem('dailyfit_logs', JSON.stringify(logs));
            updateDashboard();
        }

        function startStopwatch() {
            if(isRunning) return;
            isRunning = true;
            if("mediaSession" in navigator) navigator.mediaSession.playbackState = "playing";
            stopwatchInterval = setInterval(() => { secondsElapsed++; updateStopwatchDisplay(); }, 1000);
        }

        function pauseStopwatch() {
            isRunning = false; clearInterval(stopwatchInterval);
            if("mediaSession" in navigator) navigator.mediaSession.playbackState = "paused";
        }

        function saveStopwatch() {
            const mins = parseFloat((secondsElapsed / 60.0).toFixed(2));
            if(mins > 0) { logExercise("5", mins); alert("Đã lưu " + mins + " phút chạy bộ!"); }
            pauseStopwatch(); secondsElapsed = 0; updateStopwatchDisplay();
        }

        function updateStopwatchDisplay() {
            const m = Math.floor(secondsElapsed / 60).toString().padStart(2, '0');
            const s = (secondsElapsed % 60).toString().padStart(2, '0');
            document.getElementById('stopwatchDisplay').innerText = `${m}:${s}`;
        }

        let chartInstance = null;
        function openReport() {
            document.getElementById('reportModal').classList.remove('hidden');
            const dates = [...new Set(logs.map(l => l.date))].sort();
            const calsData = dates.map(d => logs.filter(l => l.date === d).reduce((s, l) => s + l.cals, 0));
            const ctx = document.getElementById('trendChart').getContext('2d');
            if(chartInstance) chartInstance.destroy();
            chartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: dates,
                    datasets: [{ label: 'Calo tiêu hao (kcal)', data: calsData, borderColor: '#38BDF8', backgroundColor: 'rgba(56, 189, 248, 0.1)', fill: true, tension: 0.4, borderWidth: 3 }]
                },
                options: { responsive: true, plugins: { legend: { display: false } } }
            });
            document.getElementById('webhookInput').value = localStorage.getItem('dailyfit_webhook') || '';
        }

        function closeReport() { document.getElementById('reportModal').classList.add('hidden'); }

        function syncGoogleSheets() {
            const url = document.getElementById('webhookInput').value.trim();
            if(!url) { alert("Vui lòng nhập Webhook URL!"); return; }
            localStorage.setItem('dailyfit_webhook', url);
            fetch(url, { method: 'POST', mode: 'no-cors', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(logs) })
                .then(() => alert("Đồng bộ dữ liệu thành công!"))
                .catch(err => alert("Lỗi đồng bộ: " + err));
        }

        function resetAllData() {
            if(!confirm("CẢNH BÁO: Xóa toàn bộ dữ liệu và khôi phục cài đặt gốc?")) return;
            localStorage.clear(); logs = []; updateDashboard(); closeReport(); alert("Đã reset ứng dụng thành công!");
        }

        window.onload = init;
    </script>
